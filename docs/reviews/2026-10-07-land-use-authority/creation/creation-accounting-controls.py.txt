import hashlib
import json
import os
from pathlib import Path
import subprocess

root = Path('/home/nathaniel/.local/state/openplan/land-use-creation-20261007')
app = root / 'openplan'
out = Path('/tmp/openplan-plan-creation-accounting-controls')
out.mkdir(exist_ok=True)
inventory = app / 'src/test/migrations/inventory.test.ts'
columns = app / 'src/test/a-column-nothing-reads-is-a-question.test.ts'
changelog = root / 'CHANGELOG.md'
originals = {p: p.read_bytes() for p in [inventory, columns, changelog]}
tests = ['src/test/migrations/inventory.test.ts', 'src/test/a-column-nothing-reads-is-a-question.test.ts', 'src/test/migrations/release-ordering.test.ts']
cases = [
    ('baseline', None, None, None, None),
    ('harmless', inventory, 'const EXPECTED = {', '// Creation catalog accounting control.\nconst EXPECTED = {', None),
    ('missing-relation', inventory, 'relations: 297,', 'relations: 296,', '296'),
    ('missing-table', inventory, 'tables: 283,', 'tables: 282,', '282'),
    ('missing-rls-table', inventory, 'rlsEnabledTables: 283,', 'rlsEnabledTables: 282,', '282'),
    ('undocumented-column', columns, 'land_use_plan_creation_commands.command_text', 'land_use_plan_creation_commands.nonexistent', 'land_use_plan_creation_commands.command_text'),
    ('stale-column', columns, 'land_use_plan_creation_commands.descriptor_text', 'land_use_plan_creation_commands.command_id', 'land_use_plan_creation_commands.command_id'),
    ('missing-migration-note', changelog, '20261016000007_land_use_plan_creation_commands.sql', 'unlisted creation migration', '20261016000007'),
]
results = []
try:
    for name, path, old, new, marker in cases:
        for p, data in originals.items():
            p.write_bytes(data)
        if path:
            text = path.read_text()
            assert text.count(old) == 1, (name, text.count(old))
            path.write_text(text.replace(old, new))
        run = subprocess.run(['npx', 'vitest', 'run', *tests, '--maxWorkers=1'], cwd=app, env={**os.environ, 'NODE_OPTIONS': '--max-old-space-size=6144'}, text=True, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, timeout=120)
        (out / f'{name}.log').write_text(run.stdout)
        matched = run.returncode == 0 if marker is None else run.returncode != 0 and marker in run.stdout and 'AssertionError' in run.stdout
        results.append({'case': name, 'exitCode': run.returncode, 'expectedFailure': marker is not None, 'marker': marker, 'matched': matched})
        (out / 'report.json').write_text(json.dumps({'results': results}, indent=2) + '\n')
        assert matched, name
finally:
    for p, data in originals.items():
        p.write_bytes(data)
    report = {'results': results, 'restored': all(p.read_bytes() == data for p, data in originals.items()), 'sha256': {str(p.relative_to(root)): hashlib.sha256(data).hexdigest() for p, data in originals.items()}, 'boundary': 'Static migration inventory, column naming and release note guards. Does not prove native permissions or visible workflow behavior.'}
    (out / 'report.json').write_text(json.dumps(report, indent=2) + '\n')
print(json.dumps(report))
