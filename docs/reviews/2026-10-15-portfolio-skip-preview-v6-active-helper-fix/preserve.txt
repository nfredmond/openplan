def preserve():
    actual_files={}
    for name,expected in read('preservation-files.json').items():
        f=Path(name)
        actual={'regular':f.is_file() and not f.is_symlink(),'mode':f.stat().st_mode&0o777,'sha256':sha(f)}
        p.exact({'regular':True,'mode':expected['mode'],'sha256':expected['sha256']},actual,'old-packet')
        actual_files[name]=actual
    # Existing whole-root inventory stays historical; only V4 entries are acquired here.
    frozen=read('v4-full-inventory-before.json')
    v4_roots=['/home/nathaniel/.local/state/openplan/project-portfolio-skip-preview-20261015-v4',
        '/home/nathaniel/.t3/worktrees/openplan/t3-9eeec4ad/docs/reviews/2026-10-15-portfolio-skip-preview-v4-preparation']
    for name in v4_roots:
        p.files(frozen[name],packet_inventory(Path(name)))
    v5=read('v5-packet-inventory-before.json')
    for name,expected in v5.items():
        p.files(expected,packet_inventory(Path(name)))
    web=protected_web()
    observed={'packetsSHA256':p.digest(actual_files),'v4InventorySHA256':p.digest(frozen),
        'v5PacketInventorySHA256':p.digest(v5),'oldWeb':web,'helpers':{},'helperStates':{},'databases':{}}
    ids=active_helper_identities()
    baseline=None
    if (OUT/'protected-runtime-before.json').exists():
        baseline=read('protected-runtime-before.json')
        p.exact({'mapping':True},{'mapping':type(baseline.get('activeHelperIds')) is dict},'protected-active-baseline')
        p.files(ids,baseline['activeHelperIds'])
        p.exact({'mapping':True},{'mapping':type(baseline.get('helpers')) is dict},'protected-helper-baseline')
        p.exact({'ids':sorted(ids.values())},{'ids':sorted(baseline['helpers'])},'protected-active-set')
    for identity in ids.values():
        i=inspect(identity)
        state=require_active_helper(identity,i)
        current={k:i[k] for k in ['Id','Name','Image','Config','HostConfig','Mounts']}
        actual=p.digest(current)
        if baseline is not None:
            p.exact({'sha256':baseline['helpers'][identity]},{'sha256':actual},'protected-helper')
        observed['helpers'][identity]=actual
        observed['helperStates'][identity]=state
    if baseline is not None:
        for database,expected in baseline['databases'].items():
            actual=snapshot(database)
            p.exact({'snapshot':expected},{'snapshot':actual},'protected-database')
            observed['databases'][database]=p.digest(actual)
    return observed
